cmake_minimum_required(VERSION 3.16)
project(sanity_zenoh_emulator C)
set(CMAKE_C_STANDARD 11)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(Z_FEATURE_SCOUTING 0 CACHE STRING "" FORCE)
set(Z_FEATURE_AUTO_RECONNECT 0 CACHE STRING "" FORCE)
set(Z_FEATURE_BATCHING 0 CACHE STRING "" FORCE)
if(NOT ZENOH_PICO_SOURCE)
    message(FATAL_ERROR "Set ZENOH_PICO_SOURCE to prepared Pico 1.10.1")
endif()
add_subdirectory("${ZENOH_PICO_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/pico")
add_executable(receiver_emulator receiver_emulator.c ../main/control.c)
target_include_directories(receiver_emulator PRIVATE ../main)
target_link_libraries(receiver_emulator PRIVATE zenohpico::lib pthread)
target_compile_options(receiver_emulator PRIVATE -Wall -Wextra -Werror)

add_executable(comms_emulator comms_emulator.c ../main/control.c ../main/test_config.c ../main/timing_wire.c ../main/fc_telemetry.c)
target_include_directories(comms_emulator PRIVATE ../main)
target_link_libraries(comms_emulator PRIVATE zenohpico::lib pthread)
target_compile_definitions(comms_emulator PRIVATE CONFIG_SANITY_COMMS_TEST_ONLY=1)
target_compile_options(comms_emulator PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)

if(Z_FEATURE_CONNECTIVITY)
    add_executable(session_watch_emulator session_watch_emulator.c ../main/session_watch.c)
    target_include_directories(session_watch_emulator PRIVATE ../main)
    target_link_libraries(session_watch_emulator PRIVATE zenohpico::lib pthread)
    target_compile_options(session_watch_emulator PRIVATE -Wall -Wextra -Werror)
endif()

add_executable(transport_lifetime_emulator transport_lifetime_emulator.c)
target_link_libraries(transport_lifetime_emulator PRIVATE zenohpico::lib pthread)
target_compile_options(transport_lifetime_emulator PRIVATE -Wall -Wextra -Werror)

add_executable(mocap_emulator mocap_emulator.c ../main/mocap_state.c ../main/control.c)
target_include_directories(mocap_emulator PRIVATE ../main)
target_link_libraries(mocap_emulator PRIVATE zenohpico::lib pthread m)
target_compile_options(mocap_emulator PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)
